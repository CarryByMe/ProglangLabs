`1000 & 100 = 00000000 00000000 00000011 11101000 & 00000000 00000000 00000000 01100100 = 00000000 00000000 00000000 01100000 = 2^6 + 2^5 = 64 + 32 = 96`  
`00000000 00000000 00000000 01100000 << 10 = 00000000 00000001 10000000 00000000 = 2^16 + 2^15 = 65536 + 32768 = 98304`

* `(1000 >> 4) & 15`  
`1000 >> 4 = 00000000 00000000 00000011 11101000 >> 4 = 00000000 00000000 00000000 00111110 = 2^5 + 2^4 + 2^3 + 2^2 + 2^1 = 32 + 16 + 8 + 4 + 2 = 62`  
`62 & 15 = 00000000 00000000 00000000 00111110 & 00000000 00000000 00000000 00001111 = 00000000 00000000 00000000 00001110 = 2^3 + 2^2 + 2^1 = 8 + 4 + 2 = 14`

`1234 = 00000000 00000000 00000100 11010010`
